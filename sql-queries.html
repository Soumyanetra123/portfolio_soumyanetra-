1. Multi-Table Revenue & Department Aggregation
INNER JOIN | LEFT JOIN | GROUP BY | DATE Logic

Extracts monthly revenue breakdown by consolidating patient, visit, department, and billing records.

-- Consolidate department billing totals for active inpatient visits
SELECT 
    d.DepartmentName,
    COUNT(DISTINCT v.VisitID) AS TotalVisits,
    SUM(b.InvoiceAmount) AS TotalRevenue
FROM Departments d
INNER JOIN Visits v ON d.DepartmentID = v.DepartmentID
INNER JOIN Billing b ON v.VisitID = b.VisitID
LEFT JOIN Doctors doc ON v.DoctorID = doc.DoctorID
WHERE b.BillingDate >= DATEADD(month, -1, GETDATE())
GROUP BY d.DepartmentName
ORDER BY TotalRevenue DESC;
2. Visitor Scan Frequency Enforcement (Stored Procedure)
Stored Procedures | Conditional Logic | Subqueries

Enforces scanning limits per patient UHID prior to granting visitor clearance.

-- Stored Procedure to validate daily scan allowances
CREATE PROCEDURE sp_CheckVisitorLimit
    @UHID VARCHAR(20),
    @WardType VARCHAR(20)
AS
BEGIN
    DECLARE @ScanCount INT;
    
    SELECT @ScanCount = COUNT(*) 
    FROM VisitorLogs 
    WHERE UHID = @UHID 
      AND CAST(ScanTime AS DATE) = CAST(GETDATE() AS DATE);

    IF (@WardType = 'ICU' AND @ScanCount >= 2)
        RAISERROR('ICU daily scan limit reached.', 16, 1);
    ELSE
        SELECT 'Access Granted' AS Status;
END;
