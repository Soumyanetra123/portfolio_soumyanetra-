10 Production-Grade Scripts | Multi-Table Joins, Aggregations & Date/Time Operations
1. Monthly Departmental Revenue & Visit Aggregation
Aggregates revenue, distinct visits, and average invoice values using DATEADD and DATEDIFF.
SELECT d.DepartmentName, COUNT(DISTINCT v.VisitID) AS TotalVisits, SUM(b.InvoiceAmount) AS GrossRevenue
FROM Departments d
INNER JOIN Visits v ON d.DepartmentID = v.DepartmentID
INNER JOIN Billing b ON v.VisitID = b.VisitID
WHERE b.BillingDate >= DATEADD(month, -1, GETDATE()) AND b.PaymentStatus = 'Paid'
GROUP BY d.DepartmentName ORDER BY GrossRevenue DESC;
2. ICU vs. Ward Daily Visitor Scan Audit
Audits daily visitor scanning limits per UHID across ward categories.
SELECT p.UHID, p.PatientName, w.WardType, CAST(vl.ScanTime AS DATE) AS ScanDate, COUNT(vl.LogID) AS DailyScans
FROM Patients p
INNER JOIN Admissions a ON p.UHID = a.UHID
INNER JOIN Wards w ON a.WardID = w.WardID
INNER JOIN VisitorLogs vl ON p.UHID = vl.UHID
WHERE vl.ScanTime >= DATEADD(day, -7, GETDATE())
GROUP BY p.UHID, p.PatientName, w.WardType, CAST(vl.ScanTime AS DATE)
HAVING (w.WardType = 'ICU' AND COUNT(vl.LogID) > 2) OR (w.WardType = 'General' AND COUNT(vl.LogID) > 4);
3. Doctor Consultation & Tariff Performance
Multi-table join evaluating consultant performance against standard tariff rates.
SELECT doc.DoctorID, CONCAT(doc.FirstName, ' ', doc.LastName) AS DoctorName, s.ServiceName, SUM(b.FinalAmount) AS RealizedRevenue
FROM Doctors doc
INNER JOIN Visits v ON doc.DoctorID = v.DoctorID
INNER JOIN ServiceRendered sr ON v.VisitID = sr.VisitID
INNER JOIN Services s ON sr.ServiceID = s.ServiceID
INNER JOIN Billing b ON v.VisitID = b.VisitID
WHERE v.VisitDate BETWEEN DATEADD(day, -30, GETDATE()) AND GETDATE()
GROUP BY doc.DoctorID, doc.FirstName, doc.LastName, s.ServiceName;
4. Average Length of Stay (ALOS) Analysis
Calculates patient inpatient duration metrics using DATEDIFF.
SELECT d.DepartmentName, COUNT(a.AdmissionID) AS Discharges, AVG(DATEDIFF(day, a.AdmissionDate, a.DischargeDate)) AS AvgStayDays
FROM Admissions a
INNER JOIN Wards w ON a.WardID = w.WardID
INNER JOIN Departments d ON w.DepartmentID = d.DepartmentID
WHERE a.DischargeDate IS NOT NULL AND a.DischargeDate >= DATEADD(month, -3, GETDATE())
GROUP BY d.DepartmentName ORDER BY AvgStayDays DESC;
5. Outstanding Insurance TPA Claim Breakdown
Filters pending insurance claims overdue by more than 15 days.
SELECT tpa.TPAName, p.UHID, p.PatientName, (b.InvoiceAmount - b.ClaimedAmount) AS PendingBalance
FROM Billing b
INNER JOIN Visits v ON b.VisitID = v.VisitID
INNER JOIN Patients p ON v.UHID = p.UHID
INNER JOIN TPACoverage tpa ON v.TPAID = tpa.TPAID
WHERE b.PaymentStatus = 'Pending_Insurance' AND b.BillingDate <= DATEADD(day, -15, GETDATE());
6. Emergency Lab Test Turnaround Time
Evaluates lab turnaround durations in minutes for emergency patients.
SELECT lt.TestName, COUNT(lr.RequestID) AS TotalRequested, AVG(DATEDIFF(minute, lr.RequestedTime, lr.ResultTime)) AS AvgTurnaroundMins
FROM LabRequests lr
INNER JOIN LabTests lt ON lr.TestID = lt.TestID
INNER JOIN Visits v ON lr.VisitID = v.VisitID
WHERE v.VisitType = 'Emergency' GROUP BY lt.TestName;
7. Pharmacy Inventory Expiry Alert
Identifies medications expiring within 60 days or falling below reorder limits.
SELECT m.MedicineName, s.SupplierName, i.StockQuantity, DATEDIFF(day, GETDATE(), i.ExpiryDate) AS DaysToExpiry
FROM Inventory i
INNER JOIN Medicines m ON i.MedicineID = m.MedicineID
INNER JOIN Suppliers s ON i.SupplierID = s.SupplierID
WHERE i.ExpiryDate BETWEEN GETDATE() AND DATEADD(day, 60, GETDATE());
8. 30-Day Readmission Tracking (Window Functions)
Uses LAG() to evaluate interval days between consecutive admissions per patient.
WITH PatientHistory AS (
    SELECT UHID, AdmissionDate, LAG(AdmissionDate) OVER (PARTITION BY UHID ORDER BY AdmissionDate) AS PrevAdmission
    FROM Admissions
)
SELECT p.UHID, p.PatientName, DATEDIFF(day, ph.PrevAdmission, ph.AdmissionDate) AS DaysBetween
FROM PatientHistory ph INNER JOIN Patients p ON ph.UHID = p.UHID
WHERE ph.PrevAdmission IS NOT NULL AND DATEDIFF(day, ph.PrevAdmission, ph.AdmissionDate) <= 30;
9. Doctor Quality Rating Aggregation
Calculates average patient feedback ratings by medical department.
SELECT doc.DoctorID, CONCAT(doc.FirstName, ' ', doc.LastName) AS DoctorName, ROUND(AVG(CAST(pf.Rating AS FLOAT)), 2) AS AvgRating
FROM DoctorFeedback pf
INNER JOIN Visits v ON pf.VisitID = v.VisitID
INNER JOIN Doctors doc ON v.DoctorID = doc.DoctorID
GROUP BY doc.DoctorID, doc.FirstName, doc.LastName HAVING COUNT(pf.FeedbackID) >= 10;
10. Hourly Emergency Arrival Breakdown
Extracts arrival hour trends to assist emergency room staffing plans.
SELECT DATEPART(hour, v.VisitDate) AS ArrivalHour, COUNT(v.VisitID) AS EmergencyArrivals
FROM Visits v
WHERE v.VisitType = 'Emergency' AND v.VisitDate >= DATEADD(day, -30, GETDATE())
GROUP BY DATEPART(hour, v.VisitDate) ORDER BY ArrivalHour ASC;
